# 22 — 双面板命令块「一键复制」按钮

**需求**：面板里的"一键部署命令""一键解密命令"要能直接复制。
**结论**：PHP 与 Python 两个面板的全部命令块（静态 + 动态生成）现在都带「复制」按钮，点击即入剪贴板。线上 **v2.9.6**。

---

# 一、修复前现状

两个面板的命令块统一是 `<pre class="dt-code">`：

```html
<pre class="dt-code">:: 方式1: PowerShell 下载运行（自动定位系统自带 csc.exe，无需 PATH）
powershell -w hidden -c "IWR -Uri 'http://.../jasmin_win_x.cs' -OutFile $env:TEMP\j.cs; ..."
</pre>
```

`grep -rn "复制\|copy\|clipboard"` 全仓库结果：**零个复制按钮**。Python 面板里只有三处纯文案：

```html
<div class="dt-step"><span>2</span>复制下方命令直接运行</div>
```

于是操作员只能手动拖选——而 Windows 那条命令带 `$env:TEMP`、单双引号嵌套、反斜杠，**手动拖选极易漏字符或带上换行**，正是需要一键复制的地方。

**注意**：按钮做成 `<pre>` 的**兄弟节点**（外套一层 `.dt-code-wrap`），不是子节点。
若做成子节点，`<pre>` 带 `overflow-x:auto`，横向滚动时按钮会跟着内容滑出视野。
移动端则改为把按钮压到顶部（`padding-top:34px`），避免遮挡首行。

## 2.2 JS（两份页面同步，插在 `<script>` 开头）

```js
/* 本面板多为 http（非安全上下文）→ navigator.clipboard 为 undefined，
   必须带 document.execCommand('copy') 回退，否则点击无反应。 */
function jasminCopyFallback(txt){
  var ta=document.createElement('textarea');
  ta.value=txt;ta.setAttribute('readonly','');
  ta.style.cssText='position:fixed;top:0;left:-9999px;opacity:0';
  document.body.appendChild(ta);
  ta.focus();ta.select();
  try{ta.setSelectionRange(0,txt.length);}catch(e){}
  var ok=false;
  try{ok=document.execCommand('copy');}catch(e){ok=false;}
  document.body.removeChild(ta);
  return ok;
}
function jasminDoCopy(txt,btn){
  var finish=function(ok){
    btn.textContent=ok?'✓ 已复制':'复制失败';
    btn.classList.toggle('ok',ok);btn.classList.toggle('err',!ok);
    clearTimeout(btn._jt);
    btn._jt=setTimeout(function(){btn.textContent='复制';btn.classList.remove('ok','err');},1500);
  };
  if(navigator.clipboard&&window.isSecureContext){
    navigator.clipboard.writeText(txt).then(function(){finish(true);},function(){finish(jasminCopyFallback(txt));});
  }else{
    finish(jasminCopyFallback(txt));
  }
}
function jasminEnhanceCode(root){
  root=root&&root.nodeType===1?root:document;
  var list=[],i;
  if(root.matches&&root.matches('pre.dt-code'))list.push(root);
  var found=root.querySelectorAll?root.querySelectorAll('pre.dt-code'):[];
  for(i=0;i<found.length;i++)list.push(found[i]);
  for(i=0;i<list.length;i++){
    var pre=list[i];
    if(!pre.parentNode||pre.getAttribute('data-jcopy'))continue;
    pre.setAttribute('data-jcopy','1');
    var raw=pre.textContent;              /* 先取原文再插按钮，否则会把按钮文字也复制进去 */
    var wrap=document.createElement('div');
    wrap.className='dt-code-wrap';
    pre.parentNode.insertBefore(wrap,pre);
    wrap.appendChild(pre);
    var b=document.createElement('button');
    b.type='button';b.className='copy-btn';b.textContent='复制';b.title='复制到剪贴板';
    b._jtRaw=raw;
    b.addEventListener('click',function(ev){
      ev.preventDefault();ev.stopPropagation();
      jasminDoCopy(this._jtRaw,this);
    });
    wrap.appendChild(b);
  }
}
(function(){
  function boot(){jasminEnhanceCode(document);}
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',boot);else boot();
  /* 载荷生成弹窗、一键解密展开行都是 innerHTML 动态插入 → 必须监听 */
  try{
    new MutationObserver(function(muts){
      for(var i=0;i<muts.length;i++){
        var an=muts[i].addedNodes;
        for(var j=0;j<an.length;j++)if(an[j].nodeType===1)jasminEnhanceCode(an[j]);
      }
    }).observe(document.documentElement,{childList:true,subtree:true});
  }catch(e){}
})();
```

三个设计要点：

1. **`execCommand` 回退是必需的，不是保险**。面板跑在 `http://34.2.145.178:8080`，`window.isSecureContext` 为假（实测 `undefined`），`navigator.clipboard` 也是 `undefined`。只用 `navigator.clipboard.writeText()` 会直接抛 `TypeError`——点了没有任何反应，而这是最容易漏测的一种"看起来实现完了"。
2. **先取 `textContent` 再插按钮**。顺序反了，`pre.textContent` 就会把按钮文字 `复制` 也带进去，复制出来的命令末尾多两个字——命令直接执行失败。
3. **`MutationObserver` 兜住动态块**。"一键部署命令"弹窗是 `res.innerHTML=html` 拼出来的，表格行内"一键解密"是 `tr.after(nr)` 插进去的——都在页面加载之后。只在 `DOMContentLoaded` 里扫一遍，这两处永远没有按钮。`data-jcopy` 标记防重复包裹（否则观察器会被自己的插入动作反复触发）。

## 过程记录（如实留痕）

第一次 E2E 建号后登录返回 `denied` 而非 `invalid token`——说明 CSRF 通过、`password_verify` 失败。
根因：bcrypt 哈希经 **shell 双引号**传进 `mysql -e "..."`，`$2y$12$hKj...` 里的 `$2y`、`$12`、`$hKj` 被 shell 当变量展开，落库成
`y2/yoMGAPIi0RUhctKsG..mGcbrzum0skzaMcqV9GLWa3uD61Gda`（60 字符长度都变了）。
改用 base64 传递 PHP 脚本（`echo <b64> | base64 -d | docker exec -i docker-php-1 php`）后回读校验：

```
stored_len=60 prefix=$2y$12$
回读后 password_verify = OK
误用旧坏哈希 verify = FAIL（预期）
```

失败的那次被限速器记了一笔，但登录成功后 `rl_reset()` 已自动清零，`rate_limit` 表无残留。
